def login_truck_owner(username: str, password: str) -> dict | None:
    url = f"{BASE_URL}/truck_owner/login"
    data = {"username": username, "password": password}
    try:
        response = requests.post(url, json=data)
        response.raise_for_status()
        return response.json()  # Expected to return dict with success, token, etc.
    except requests.RequestException as e:
        print(f"Truck owner login failed: {e}")
        return None

def login_contractor(username: str, password: str) -> dict | None:
    url = f"{BASE_URL}/contractor/login"
    data = {"username": username, "password": password}
    try:
        response = requests.post(url, json=data)
        response.raise_for_status()
        return response.json()
    except requests.RequestException as e:
        print(f"Contractor login failed: {e}")
        return None

def logout(token: str) -> bool:
    url = f"{BASE_URL}/logout"
    headers = {"Authorization": f"Bearer {token}"}
    try:
        response = requests.post(url, headers=headers)
        response.raise_for_status()
        return True
    except requests.RequestException as e:
        print(f"Logout failed: {e}")
        return False

# Dependency for DB session
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


# Create a new job
@router.post("/", response_model=JobOut, status_code=status.HTTP_201_CREATED)
def JobCreate(job: JobCreate, db: Session = Depends(get_db)):
    new_job = Job(
        description=job.description,
        status=JobStatus.pending
    )
    db.add(new_job)
    db.commit()
    db.refresh(new_job)
    return new_job


# Upload a slip file and mark job as completed
@router.post("/{job_id}/upload-slip")
def upload_slip(job_id: int, file: UploadFile = File(...), db: Session = Depends(get_db)):
    job = db.query(Job).filter(Job.id == job_id).first()
    if not job:
        raise HTTPException(status_code=404, detail="Job not found")

    # Create folder if not exists
    upload_dir = "backend/uploads/job_slips"
    os.makedirs(upload_dir, exist_ok=True)

    # Save the uploaded file
    file_path = os.path.join(upload_dir, file.filename)
    with open(file_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)

    # Update job record
    job.slip = file_path
    job.status = JobStatus.completed
    db.commit()

    return {"message": "Slip uploaded successfully"}

from sqlalchemy import Column, Integer, String, Enum, ForeignKey
from sqlalchemy.orm import relationship
import enum




class JobStatus(enum.Enum):
    pending = "pending"
    active = "active"
    completed = "completed"


class Job(Base):
    __tablename__ = "jobs"

    id = Column(Integer, primary_key=True, index=True)
    title = Column(String, nullable=False)
    status = Column(Enum(JobStatus), default=JobStatus.pending)

    user_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    user = relationship("User", back_populates="jobs")
